export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } const { imageUrl, prompt, duration } = req.body; if (!imageUrl) { return res.status(400).json({ error: 'Image is required' }); } const FAL_API_KEY = process.env.FAL_API_KEY; if (!FAL_API_KEY) { return res.status(500).json({ error: 'FAL_API_KEY not configured on server.' }); } try { const submitRes = await fetch('https://queue.fal.run/fal-ai/kling-video/v1.6/standard/image-to-video', { method: 'POST', headers: { 'Authorization': `Key ${FAL_API_KEY}`, 'Content-Type': 'application/json', }, body: JSON.stringify({ image_url: imageUrl, prompt: prompt || 'cinematic motion, smooth animation', duration: duration || '5', aspect_ratio: '16:9', }), }); if (!submitRes.ok) { const err = await submitRes.text(); return res.status(submitRes.status).json({ error: err }); } const submitData = await submitRes.json(); const requestId = submitData.request_id; let attempts = 0; while (attempts < 40) { await new Promise(r => setTimeout(r, 3000)); attempts++; const statusRes = await fetch( `https://queue.fal.run/fal-ai/kling-video/v1.6/standard/image-to-video/requests/${requestId}`, { headers: { 'Authorization': `Key ${FAL_API_KEY}` } } ); const statusData = await statusRes.json(); if (statusData.status === 'COMPLETED') { const videoUrl = statusData.output?.video?.url; if (videoUrl) return res.status(200).json({ video: videoUrl }); } if (statusData.status === 'FAILED') { return res.status(500).json({ error: 'Video generation failed.' }); } } return res.status(504).json({ error: 'Timeout.' }); } catch (err) { console.error(err); return res.status(500).json({ error: 'Internal server error.' }); } }
